/* Hand-written linker symbols that splat's undefined_syms_auto.txt can't produce. */

/* render/particle (func_00380518, func_00381310): the VU1 microprogram entry at VU address 0, named by
   config/reloc_addrs.txt. splat only lists symbols its reference tracking marks as referenced, which a
   reloc_addrs-only name isn't, so it never appears in undefined_syms_auto.txt. */
D_0 = 0x0;

/* render/bezierman (func_0038AF30): the two UV grids (6x6 and 8x8 {float u, v}) in .bss. The target loads their
   addresses with a lui in a branch delay slot that splat does not pair with its addiu, so it emits raw immediates
   and never names the symbols. */
D_00504A00 = 0x504A00;
D_00504B80 = 0x504B80;

/* ai/control/handplantcontrol (func_0013AF28): the function-local static's init guard, one word after the static

   as its own scalar global. */
D_004A4C04 = 0x4A4C04;

/* ai/control/handplantcontrol (func_0013FB20): an 8-byte index object {id:8, val:24} in the gp region, addressed
   by splat only as D_004A4C00 + 8; the static initialiser stores it as its own object. */
D_004A4C08 = 0x4A4C08;
